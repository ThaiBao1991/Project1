1. Chỉnh chế độ IME để access không tự động đổi font

Cách thực hiện lại để đảm bảo bạn làm đúng:

  1. Nhấn Alt + F11 để mở cửa sổ VBA.
  2. Chọn menu Insert > Module.
  3. Dán đoạn mã này vào (đây là bản đã kiểm tra kỹ để chạy trên mọi bảng):
Sub FixAllTablesIMEMode()
    Dim db As DAO.Database
    Dim tdf As DAO.TableDef
    Dim fld As DAO.Field
    Dim prp As DAO.Property
    Dim count As Long
    
    Set db = CurrentDb
    count = 0
    
    ' Quét qua tất cả các bảng trong database
    For Each tdf In db.TableDefs
        ' Bỏ qua các bảng hệ thống của Access (bắt đầu bằng MSys)
        If Left(tdf.Name, 4) <> "MSys" Then
            For Each fld In tdf.Fields
                On Error Resume Next
                ' Kiểm tra xem trường có thuộc tính ImeMode không
                ' Nếu có, gán giá trị 0 (tương ứng với "No Control")
                fld.Properties("ImeMode") = 0
                If Err.Number = 0 Then
                    count = count + 1
                End If
                On Error GoTo 0
            Next fld
        End If
    Next tdf
    
    MsgBox "Hoàn tất! Đã cập nhật thuộc tính IME cho " & count & " trường trong tất cả các bảng.", vbInformation
End Sub

3. Tạo dữ liệu bổ sung các ô cho bảng với điều kiện

Sub BoSungTruongChoBang()
    Dim db As DAO.Database
    Dim tdf As DAO.TableDef
    Dim fld As DAO.Field
    Dim fldCheck As DAO.Field
    Dim hasField As Boolean
    Dim countTable As Long
    
    Set db = CurrentDb
    countTable = 0
    
    ' Duyet qua toan bo cac bang trong co so du lieu
    For Each tdf In db.TableDefs
        ' Bo qua cac bang he thong cua Access
        If Left(tdf.Name, 4) <> "MSys" Then
            
            ' --- DIEU KIEN MOI: Ten bang phai chua chu "diem_kiem" (chi chap nhan chu thuong, co hoa la bo qua) ---
            ' InStr(1, tdf.Name, "diem_kiem", vbBinaryCompare) se tim kiem phan biet chu hoa/thuong chinh xac
            If InStr(1, tdf.Name, "diem_kiem", vbBinaryCompare) > 0 Then
                
                hasField = False
                
                ' Kiem tra xem bang co chua field "Hang muc yeu cau" hay khong
                On Error Resume Next
                Set fldCheck = tdf.Fields("Hang muc yeu cau")
                If Err.Number = 0 Then
                    hasField = True
                End If
                Err.Clear
                On Error GoTo 0
                
                ' Neu co chua field "Hang muc yeu cau", tien hanh bo sung cac field phia sau
                If hasField Then
                    ' 1. "Tinh trang su dung"
                    If Not FieldExists(tdf, "Tinh trang su dung") Then
                        Set fld = tdf.CreateField("Tinh trang su dung", dbText, 50)
                        tdf.Fields.Append fld
                    End If
                    
                    ' 2. "Tinh trang chi tiet"
                    If Not FieldExists(tdf, "Tinh trang chi tiet") Then
                        Set fld = tdf.CreateField("Tinh trang chi tiet", dbText, 100)
                        tdf.Fields.Append fld
                    End If
                    
                    ' 3. "Pham vi su dung"
                    If Not FieldExists(tdf, "Pham vi su dung") Then
                        Set fld = tdf.CreateField("Pham vi su dung", dbText, 100)
                        tdf.Fields.Append fld
                    End If
                    
                    ' 4. "Phi hieu chinh"
                    If Not FieldExists(tdf, "Phi hieu chinh") Then
                        Set fld = tdf.CreateField("Phi hieu chinh", dbCurrency)
                        tdf.Fields.Append fld
                    End If
                    
                    ' 5. "Don vi tien"
                    If Not FieldExists(tdf, "Don vi tien") Then
                        Set fld = tdf.CreateField("Don vi tien", dbText, 20)
                        tdf.Fields.Append fld
                    End If
                    
                    ' 6. "SMI Phan Dinh"
                    If Not FieldExists(tdf, "SMI Phan Dinh") Then
                        Set fld = tdf.CreateField("SMI Phan Dinh", dbText, 100)
                        tdf.Fields.Append fld
                    End If
                    
                    ' 7. "SMI Tinh Trang Chi Tiet"
                    If Not FieldExists(tdf, "SMI Tinh Trang Chi Tiet") Then
                        Set fld = tdf.CreateField("SMI Tinh Trang Chi Tiet", dbText, 100)
                        tdf.Fields.Append fld
                    End If
                    
                    countTable = countTable + 1
                End If
                
            End If ' Ket thuc dieu kien kiem tra ten bang chua "diem_kiem"
            
        End If
    Next tdf
    
    MsgBox "Hoan tat! Da bo sung cac truong thanh cong cho " & countTable & " bang hop le.", vbInformation
End Sub


' Ham phu tro kiem tra xem field da ton tai trong bang chua de tranh loi tao trung
﻿Function FieldExists(tdf As DAO.TableDef, fldName As String) As Boolean
    Dim f As DAO.Field
    FieldExists = False
    For Each f In tdf.Fields
        If LCase(f.Name) = LCase(fldName) Then
            FieldExists = True
            Exit Function
        End If
    Next f
End Function
